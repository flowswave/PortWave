cmake_minimum_required(VERSION 3.25)
project(PortWave VERSION 0.1.0 LANGUAGES CXX)
if (WIN32)
  # Resource compiler: src/app/PortWave.rc embeds the application icon.
  enable_language(RC)
endif()
include(cmake/MsvcDependencies.cmake)
include(CTest)
option(PORTWAVE_BUILD_GUI "Build the Qt desktop application" ON)
option(PORTWAVE_BENCHMARKS "Build measured core benchmark" OFF)
add_library(portwave_options INTERFACE)
target_compile_features(portwave_options INTERFACE cxx_std_20)
if(MSVC)
  target_compile_options(portwave_options INTERFACE /W4 /permissive- /utf-8 /Zc:__cplusplus)
else()
  target_compile_options(portwave_options INTERFACE -Wall -Wextra -Wpedantic)
endif()
add_subdirectory(src/core)
if(PORTWAVE_BUILD_GUI)
  cmake_policy(PUSH)
  cmake_policy(SET CMP0077 NEW)
  set(BUILD_EXAMPLES OFF)
  set(BUILD_TESTS OFF)
  set(INSTALL_PROJECT OFF)
  add_subdirectory(third_party/miniz-3.1.2 EXCLUDE_FROM_ALL)
  cmake_policy(POP)
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Network SerialPort Concurrent PrintSupport Svg Test)
  set(CMAKE_AUTOMOC ON)
  set(CMAKE_AUTORCC ON)
  include(cmake/QScintilla.cmake)
  add_subdirectory(src/infrastructure)
  add_subdirectory(src/services)
  add_subdirectory(src/editor)
  add_subdirectory(src/features)
  add_subdirectory(src/ui)
  add_subdirectory(src/app)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
if(PORTWAVE_BENCHMARKS)
  add_subdirectory(benchmarks)
endif()
